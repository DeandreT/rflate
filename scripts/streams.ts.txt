//@@DEFLATE
/**
 * Streaming DEFLATE compression
 */
export class Deflate {
  /**
   * Creates a DEFLATE stream
   * @param opts The compression options
   * @param cb The callback to call whenever data is deflated
   */
  constructor(opts: DeflateOptions, cb?: FlateStreamHandler);
  /**
   * Creates a DEFLATE stream
   * @param cb The callback to call whenever data is deflated
   */
  constructor(cb?: FlateStreamHandler);
  constructor(opts?: DeflateOptions | FlateStreamHandler, cb?: FlateStreamHandler) {
    if (typeof opts == 'function') cb = opts as FlateStreamHandler, opts = {};
    this.ondata = cb;
    const o = this.o = (opts as DeflateOptions) || {};
    this.y = o.dictionary ? slc(o.dictionary.subarray(-32768), 0) : et;
    this.h = 0;
    this.l = 0;
  }
  private o: DeflateOptions;
  // dictionary (copied)
  private y: Uint8Array;
  // WebAssembly stream handle
  private h: number;
  // finished
  private l: number;
  // checksum of the input once finished
  private k: number;
  /**
   * The handler to call whenever data is available
   */
  ondata: FlateStreamHandler;

  // compresses a chunk. f: 0 none, 1 block flush, 2 sync flush, 3 finish
  // m: checksum mode. Returns null when no output is available yet.
  private q(c: Uint8Array, f: number, m: number, pre: number, post: number) {
    if (!this.ondata) err(5);
    if (this.l) err(4);
    const x = wx();
    if (!this.h) {
      const o = this.o;
      wput(x, this.y);
      this.h = x.def_new(o.level == null ? 6 : o.level, o.mem == null ? -1 : o.mem, m, this.y.length) >>> 0;
      wreg(this, this.h, 0);
    }
    // large chunks go through in slices to bound WebAssembly memory growth
    const parts: Uint8Array[] = [];
    let n = 0, r: Uint8Array<ArrayBuffer> = null;
    for (let i = 0, e = 0; ; i = e) {
      e = Math.min(i + 1048576, c.length);
      const last = e == c.length;
      wput(x, c.subarray(i, e));
      const p = x.def_push(this.h, e - i, last ? f : 0) >>> 0, k = x.out_len();
      if (!p) wer(x);
      if (last && !parts.length) {
        if (k || f > 1) r = wget(x, p, pre, post);
        break;
      }
      if (k) parts.push(new u8(x.memory.buffer, p, k).slice()), n += k;
      if (last) {
        r = new u8(pre + n + post);
        for (let j = 0, w = pre; j < parts.length; w += parts[j++].length) r.set(parts[j], w);
        break;
      }
    }
    if (f == 3) {
      this.k = x.def_check(this.h);
      x.def_free(this.h), wunreg(this);
      this.h = 0, this.l = 1;
      wrel(x);
    }
    return r;
  }

  /**
   * Pushes a chunk to be deflated
   * @param chunk The chunk to push
   * @param final Whether this is the last chunk
   */
  push(chunk: Uint8Array, final?: boolean) {
    const r = this.q(chunk, final ? 3 : 0, 0, 0, 0);
    if (r) this.ondata(r, !!final);
  }

  /**
   * Flushes buffered uncompressed data. Useful to immediately retrieve the
   * deflated output for small inputs.
   * @param sync Whether to flush to a byte boundary. A sync flush takes 4-5
   *             extra bytes, but guarantees all pushed data is immediately
   *             decompressible. A separate DEFLATE stream may be concatenated
   *             with the current output after a sync flush.
   */
  flush(sync?: boolean) {
    const r = this.q(et, sync ? 2 : 1, 0, 0, 0);
    if (r) this.ondata(r, false);
  }
}
//@@DEFLATESYNC
export function deflateSync(data: Uint8Array, opts?: DeflateOptions) {
  return dopt(data, opts || {}, 0, 0);
}
//@@INFLATE
/**
 * Streaming DEFLATE decompression
 */
export class Inflate {
  // WebAssembly stream handle
  private h: number;
  // dictionary
  private x: Uint8Array;
  // done (final chunk pushed)
  private d: boolean;
  // DEFLATE stream ended
  private f: number;
  // unconsumed bytes after the end of the stream in the last chunk
  private t: number;
  // bytes pushed into the current stream
  private n: number;
  /**
   * The handler to call whenever data is available
   */
  ondata: FlateStreamHandler;

  /**
   * Creates a DEFLATE decompression stream
   * @param opts The decompression options
   * @param cb The callback to call whenever data is inflated
   */
  constructor(opts: InflateStreamOptions, cb?: FlateStreamHandler);
  /**
   * Creates a DEFLATE decompression stream
   * @param cb The callback to call whenever data is inflated
   */
  constructor(cb?: FlateStreamHandler);
  constructor(opts?: InflateStreamOptions | FlateStreamHandler, cb?: FlateStreamHandler) {
    // no StrmOpt here to avoid adding to workerizer
    if (typeof opts == 'function') cb = opts as FlateStreamHandler, opts = {};
    this.ondata = cb;
    const dict = opts && (opts as InflateStreamOptions).dictionary;
    this.x = dict ? slc(dict.subarray(-32768), 0) : et;
    this.h = this.f = this.t = this.n = 0;
  }

  // inflates a chunk of raw DEFLATE data
  private r(c: Uint8Array, final: boolean | number) {
    if (this.f) {
      this.t = c.length;
      return new u8(0);
    }
    const x = wx();
    if (!this.h) {
      wput(x, this.x);
      this.h = x.inf_new(this.x.length) >>> 0;
      wreg(this, this.h, 1);
    }
    // large chunks go through in slices to bound WebAssembly memory growth
    const parts: Uint8Array<ArrayBuffer>[] = [];
    for (let i = 0, e = 0; ; i = e) {
      e = Math.min(i + 1048576, c.length);
      const last = e == c.length;
      this.n += e - i;
      wput(x, c.subarray(i, e));
      const p = x.inf_push(this.h, e - i, last && final && this.n ? 1 : 0) >>> 0;
      if (!p) {
        const er = x.err();
        ifr(this);
        err(er, er == 99 && 'out of memory');
      }
      parts.push(wget(x, p, 0, 0));
      if (x.inf_ended(this.h)) {
        this.f = 1, this.t = c.length - i - x.consumed();
        break;
      }
      if (last) {
        this.t = 0;
        break;
      }
    }
    if (this.f || final) ifr(this);
    if (parts.length == 1) return parts[0];
    let n = 0;
    for (let j = 0; j < parts.length; ++j) n += parts[j].length;
    const o = new u8(n);
    for (let j = 0, w = 0; j < parts.length; w += parts[j++].length) o.set(parts[j], w);
    return o;
  }

  /**
   * Pushes a chunk to be inflated
   * @param chunk The chunk to push
   * @param final Whether this is the final chunk
   */
  push(chunk: Uint8Array, final?: boolean) {
    if (!this.ondata) err(5);
    if (this.d) err(4);
    this.d = !!final;
    this.ondata(this.r(chunk, final), !!final);
  }
}
//@@INFLATESYNC
export function inflateSync(data: Uint8Array, opts?: InflateOptions) {
  return inflt(data, opts && opts.out, opts && opts.dictionary);
}
//@@GZIP
/**
 * Streaming GZIP compression
 */
export class Gzip {
  private v = 1;
  private z = 0;
  private o: GzipOptions;
  private k: number;
  /**
   * The handler to call whenever data is available
   */
  ondata: FlateStreamHandler;

  /**
   * Creates a GZIP stream
   * @param opts The compression options
   * @param cb The callback to call whenever data is deflated
   */
  constructor(opts: GzipOptions, cb?: FlateStreamHandler);
  /**
   * Creates a GZIP stream
   * @param cb The callback to call whenever data is deflated
   */
  constructor(cb?: FlateStreamHandler);
  constructor(opts?: GzipOptions | FlateStreamHandler, cb?: FlateStreamHandler) {
    Deflate.call(this, opts, cb);
  }

  /**
   * Pushes a chunk to be GZIPped
   * @param chunk The chunk to push
   * @param final Whether this is the last chunk
   */
  push(chunk: Uint8Array, final?: boolean) {
    this.p(chunk, final ? 3 : 0);
    this.z += chunk.length;
  }

  private p(c: Uint8Array, f: number) {
    const o = this.o;
    const raw = (Deflate.prototype as unknown as { q: (c: Uint8Array, f: number, m: number, pre: number, post: number) => Uint8Array }).q.call(this, c, f, 1, this.v ? gzhl(o) : 0, f == 3 ? 8 : 0);
    if (raw) {
      if (this.v) gzh(raw, o), this.v = 0;
      if (f == 3) wbytes(raw, raw.length - 8, this.k), wbytes(raw, raw.length - 4, this.z + c.length);
      this.ondata(raw, f == 3);
    }
  }

  /**
   * Flushes buffered uncompressed data. Useful to immediately retrieve the
   * GZIPped output for small inputs.
   * @param sync Whether to flush to a byte boundary. A sync flush takes 4-5
   *             extra bytes, but guarantees all pushed data is immediately
   *             decompressible.
   */
  flush(sync?: boolean) {
    this.p(et, sync ? 2 : 1);
  }
}
//@@GZIPSYNC
export function gzipSync(data: Uint8Array, opts?: GzipOptions) {
  if (!opts) opts = {};
  const l = data.length;
  const ck = [0], d = dopt(data, opts, gzhl(opts), 8, 1, ck), s = d.length;
  return gzh(d, opts), wbytes(d, s - 8, ck[0]), wbytes(d, s - 4, l), d;
}
//@@GUNZIP
/**
 * Streaming single or multi-member GZIP decompression
 */
export class Gunzip {
  // awaiting a member header
  private v = 1;
  // total bytes pushed
  private r = 0;
  // members started
  private m = 0;
  // pending bytes
  private p: Uint8Array;
  private d: boolean;
  private f: number;
  private t: number;
  private n: number;
  /**
   * The handler to call whenever data is available
   */
  ondata: FlateStreamHandler;
  /**
   * The handler to call whenever a new GZIP member is found
   */
  onmember?: GunzipMemberHandler;

  /**
   * Creates a GUNZIP stream
   * @param opts The decompression options
   * @param cb The callback to call whenever data is inflated
   */
  constructor(opts: GunzipStreamOptions, cb?: FlateStreamHandler);
  /**
   * Creates a GUNZIP stream
   * @param cb The callback to call whenever data is inflated
   */
  constructor(cb?: FlateStreamHandler);
  constructor(opts?: GunzipStreamOptions | FlateStreamHandler, cb?: FlateStreamHandler) {
    Inflate.call(this, opts, cb);
    this.p = et;
  }

  /**
   * Pushes a chunk to be GUNZIPped
   * @param chunk The chunk to push
   * @param final Whether this is the last chunk
   */
  push(chunk: Uint8Array, final?: boolean) {
    if (!this.ondata) err(5);
    if (this.d) err(4);
    this.d = !!final;
    this.r += chunk.length;
    let p = this.p.length ? cat(this.p, chunk) : chunk, pend: Uint8Array<ArrayBuffer> = null;
    // the newest output is held back so the last one can carry the final flag
    const emit = (o: Uint8Array<ArrayBuffer>) => {
      const q = pend;
      pend = o;
      if (q && q.length) {
        // if the handler throws, stop delivering, as fflate does
        try { this.ondata(q, false); } catch (e) { pend = null; throw e; }
      }
    };
    try {
      for (;;) {
        if (this.v) {
          if (!p.length) break;
          const s = p.length > 3 ? gzs(p) : 4;
          if (s > p.length) {
            if (final) p = et;
            break;
          }
          if (this.m++ && this.onmember) emit(null), this.onmember(this.r - p.length);
          p = p.subarray(s);
          this.v = this.f = this.n = 0;
        }
        if (this.f) {
          // skip the CRC32 and ISIZE trailer, then look for another member
          if (p.length < 8) break;
          p = p.subarray(8);
          this.v = 1;
          continue;
        }
        if (!p.length) break;
        emit((Inflate.prototype as unknown as { r: (c: Uint8Array, f: number) => Uint8Array<ArrayBuffer> }).r.call(this, p, 0));
        p = p.subarray(p.length - this.t);
        if (!this.f) break;
      }
      if (final) {
        const unfinished = !this.v && !this.f && this.n;
        ifr(this);
        if (unfinished) err(0);
      }
    } catch (e) {
      // deliver what was decoded before the error, as fflate does
      emit(null);
      throw e;
    }
    this.p = final ? et : p.length ? slc(p, 0) : et;
    if (final) this.ondata(pend || new u8(0), true);
    else if (pend && pend.length) this.ondata(pend, false);
  }
}
//@@GUNZIPSYNC
export function gunzipSync(data: Uint8Array, opts?: GunzipOptions) {
  const st = gzs(data);
  if (st + 8 > data.length) err(6, 'invalid gzip data');
  return inflt(data.subarray(st, -8), opts && opts.out, opts && opts.dictionary, gzl(data));
}

/**

 */
export class Zlib {
  private v = 1;
  private o: ZlibOptions;
  private k: number;
  /**
   * The handler to call whenever data is available
   */

   * Pushes a chunk to be zlibbed

    this.p(chunk, final ? 3 : 0);
  }

  private p(c: Uint8Array, f: number) {
    const o = this.o;

    if (raw) {
      if (this.v) zlh(raw, o), this.v = 0;
      if (f == 3) wbytes(raw, raw.length - 4, bsw(this.k));
      this.ondata(raw, f == 3);
    }
  }

  /**
   * Flushes buffered uncompressed data. Useful to immediately retrieve the

   * @param sync Whether to flush to a byte boundary. A sync flush takes 4-5
   *             extra bytes, but guarantees all pushed data is immediately
   *             decompressible.
   */

    this.p(et, sync ? 2 : 1);
  }
}
//@@ZLIBSYNC
export function zlibSync(data: Uint8Array, opts?: ZlibOptions) {
  if (!opts) opts = {};
  const ck = [0], d = dopt(data, opts, opts.dictionary ? 6 : 2, 4, 2, ck);
  return zlh(d, opts), wbytes(d, d.length - 4, bsw(ck[0])), d;
}
//@@UNZLIB
/**
 * Streaming Zlib decompression
 */
export class Unzlib {
  // awaiting header (1, or 2 when a dictionary is expected)
  private v: number;
  // pending header bytes
  private p: Uint8Array;
  // trailing bytes after the DEFLATE stream
  private z: number;
  private d: boolean;
  private f: number;
  private t: number;
  /**
   * The handler to call whenever data is available
   */
  ondata: FlateStreamHandler;

  /**
   * Creates a Zlib decompression stream
   * @param opts The decompression options
   * @param cb The callback to call whenever data is inflated
   */
  constructor(opts: UnzlibStreamOptions, cb?: FlateStreamHandler);
  /**
   * Creates a Zlib decompression stream
   * @param cb The callback to call whenever data is inflated
   */
  constructor(cb?: FlateStreamHandler);
  constructor(opts?: UnzlibStreamOptions | FlateStreamHandler, cb?: FlateStreamHandler) {
    Inflate.call(this, opts, cb);
    this.v = opts && (opts as UnzlibStreamOptions).dictionary ? 2 : 1;
    this.p = et;
    this.z = 0;
  }

  /**
   * Pushes a chunk to be unzlibbed
   * @param chunk The chunk to push
   * @param final Whether this is the last chunk
   */
  push(chunk: Uint8Array, final?: boolean) {
    if (!this.ondata) err(5);
    if (this.d) err(4);
    let p = chunk;
    if (this.v) {
      p = this.p.length ? cat(this.p, chunk) : chunk;
      if (p.length < 6 && !final) {
        this.p = slc(p, 0);
        return;
      }
      p = p.subarray(zls(p, this.v - 1)), this.v = 0, this.p = et;
    }
    this.d = !!final;
    const o = (Inflate.prototype as unknown as { r: (c: Uint8Array, f: boolean) => Uint8Array<ArrayBuffer> }).r.call(this, p, final);
    if (this.f) this.z += this.t;
    if (final && this.z < 4) err(6, 'invalid zlib data');
    this.ondata(o, !!final);
  }
}
//@@UNZLIBSYNC
export function unzlibSync(data: Uint8Array, opts?: UnzlibOptions) {
  return inflt(data.subarray(zls(data, opts && opts.dictionary), -4), opts && opts.out, opts && opts.dictionary);
}
